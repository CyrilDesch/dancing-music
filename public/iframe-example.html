📖 Comment utiliser cette API

// 1. Écouter quand l'iframe est prête
window.addEventListener('message', (event) => {
  if (event.data.type === 'ready') {
    console.log('Iframe prête!');
  }
});

// 2. Envoyer une URL audio (URL distante, blob URL, ou data URL)
iframe.contentWindow.postMessage({
  type: 'setAudioUrl',
  url: 'https://example.com/music.mp3'
}, '*');

// 3. Pour un fichier local, créer un blob URL
const file = inputElement.files[0];
const blobUrl = URL.createObjectURL(file);
iframe.contentWindow.postMessage({
  type: 'setAudioUrl',
  url: blobUrl
}, '*');

// 4. Démarrer/arrêter l'audio
iframe.contentWindow.postMessage({ type: 'startAudio' }, '*');
iframe.contentWindow.postMessage({ type: 'stopAudio' }, '*');

// 5. Régler le volume (0-1)
iframe.contentWindow.postMessage({
  type: 'setVolume',
  volume: 0.5
}, '*');

// 6. Demander le statut
iframe.contentWindow.postMessage({ type: 'getStatus' }, '*');
